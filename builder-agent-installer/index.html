#!/bin/bash # # Builder Agent Installer # https://github.com/Banyan-Intelligence/builder-agent # # Usage: curl -fsSL https://banyan-intelligence.github.io/banyanintelligence-apps/builder-agent | bash # set -e # Colors for terminal output RED='\033[0;31m' GREEN='\033[0;32m' YELLOW='\033[1;33m' BLUE='\033[0;34m' CYAN='\033[0;36m' BOLD='\033[1m' NC='\033[0m' # No Color # Installation directory INSTALL_DIR="$HOME/.banyan/apps" # GitHub repository REPO="Banyan-Intelligence/builder-agent" # Print functions print_banner() { echo "" echo -e "${CYAN}${BOLD}" echo " ____ _ _ _ _ _ " echo " | __ ) _ _(_) | __| | ___ _ __ / \\ __ _ ___ _ __ | |_ " echo " | _ \\| | | | | |/ _\` |/ _ \\ '__| / _ \\ / _\` |/ _ \\ '_ \\| __|" echo " | |_) | |_| | | | (_| | __/ | / ___ \\ (_| | __/ | | | |_ " echo " |____/ \\__,_|_|_|\\__,_|\\___|_| /_/ \\_\\__, |\\___|_| |_|\\__|" echo " |___/ " echo -e "${NC}" echo -e "${BLUE} Banyan Intelligence Builder Agent Installer${NC}" echo "" } print_step() { echo -e "${BLUE}[*]${NC} $1" } print_success() { echo -e "${GREEN}[+]${NC} $1" } print_warning() { echo -e "${YELLOW}[!]${NC} $1" } print_error() { echo -e "${RED}[x]${NC} $1" } # Detect OS detect_os() { local os os=$(uname -s | tr '[:upper:]' '[:lower:]') case "$os" in darwin) echo "darwin" ;; linux) echo "linux" ;; mingw*|msys*|cygwin*) echo "windows" ;; *) print_error "Unsupported operating system: $os" exit 1 ;; esac } # Detect architecture detect_arch() { local arch arch=$(uname -m) case "$arch" in x86_64|amd64) echo "amd64" ;; arm64|aarch64) echo "arm64" ;; *) print_error "Unsupported architecture: $arch" exit 1 ;; esac } # Check for required commands check_requirements() { local missing=() for cmd in curl tar; do if ! command -v "$cmd" &> /dev/null; then missing+=("$cmd") fi done if [ ${#missing[@]} -ne 0 ]; then print_error "Missing required commands: ${missing[*]}" exit 1 fi } # Get latest version from GitHub get_latest_version() { local api_url="https://api.github.com/repos/${REPO}/releases/latest" local response response=$(curl -fsSL "$api_url" 2>/dev/null) || { # Try fetching tags if releases endpoint fails api_url="https://api.github.com/repos/${REPO}/tags" response=$(curl -fsSL "$api_url" 2>/dev/null) || { print_error "Failed to fetch version information from GitHub" exit 1 } # Extract first tag name from array echo "$response" | grep -o '"name": *"[^"]*"' | head -1 | sed 's/"name": *"//;s/"//' return } # Extract tag_name from release echo "$response" | grep -o '"tag_name": *"[^"]*"' | sed 's/"tag_name": *"//;s/"//' } # Main installation function main() { print_banner # Check requirements print_step "Checking requirements..." check_requirements print_success "All requirements satisfied" # Detect OS and architecture print_step "Detecting system information..." local os arch os=$(detect_os) arch=$(detect_arch) print_success "Detected: ${os} (${arch})" # Create installation directory print_step "Creating installation directory..." if [ ! -d "$INSTALL_DIR" ]; then mkdir -p "$INSTALL_DIR" print_success "Created ${INSTALL_DIR}" else print_success "Directory already exists: ${INSTALL_DIR}" fi # Get latest version print_step "Fetching latest version..." local tag version tag=$(get_latest_version) if [ -z "$tag" ]; then print_error "Could not determine latest version" exit 1 fi # Remove 'v' prefix for version number version="${tag#v}" print_success "Latest version: ${version}" # Construct download URL local filename="builder-agent-${version}-${os}-${arch}.tar.gz" local download_url="https://github.com/${REPO}/releases/download/${tag}/${filename}" local download_path="${INSTALL_DIR}/${filename}" # Download the release print_step "Downloading builder-agent..." echo -e " ${CYAN}URL: ${download_url}${NC}" if ! curl -fsSL -o "$download_path" "$download_url"; then print_error "Failed to download builder-agent" print_error "URL: ${download_url}" exit 1 fi print_success "Download complete" # Extract the archive print_step "Extracting archive..." if ! tar -xzf "$download_path" -C "$INSTALL_DIR"; then print_error "Failed to extract archive" rm -f "$download_path" exit 1 fi print_success "Extraction complete" # Clean up the archive print_step "Cleaning up..." rm -f "$download_path" print_success "Removed temporary files" # Make binary executable if [ -f "${INSTALL_DIR}/builder-agent" ]; then chmod +x "${INSTALL_DIR}/builder-agent" fi # Check environment variable echo "" if [ -z "${BANYAN_APPS:-}" ]; then print_warning "BANYAN_APPS environment variable is not set" echo "" echo -e "${YELLOW}To complete the installation, add the following to your shell configuration:${NC}" echo "" echo -e "${BOLD}For ~/.zshrc or ~/.bashrc:${NC}" echo "" echo -e "${CYAN}# banyan builder agent" echo 'export BANYAN_APPS="$HOME/.banyan/apps"' echo 'export PATH="$BANYAN_APPS:$PATH"' echo -e "${NC}" echo -e "Then run: ${BOLD}source ~/.zshrc${NC} (or ${BOLD}source ~/.bashrc${NC})" echo "" fi # Success message echo "" echo -e "${GREEN}${BOLD}========================================${NC}" echo -e "${GREEN}${BOLD} Builder Agent installed successfully! ${NC}" echo -e "${GREEN}${BOLD}========================================${NC}" echo "" echo -e " ${CYAN}Installation path:${NC} ${INSTALL_DIR}/builder-agent" echo -e " ${CYAN}Version:${NC} ${version}" echo "" if [ -n "${BANYAN_APPS:-}" ]; then echo -e " Run ${BOLD}builder-agent --help${NC} to get started." else echo -e " After updating your shell configuration, run ${BOLD}builder-agent --help${NC} to get started." fi echo "" } # Run main function main
